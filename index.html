export default async function handler(req, res) { res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type'); if (req.method === 'OPTIONS') { return res.status(200).end(); } const targetUrl = req.query.url || (req.body && req.body.url); if (!targetUrl) { return res.status(400).json({ success: false, error: 'Insira um link válido.' }); } // 1. Extrator dedicado para TikTok e Douyin (Alta velocidade e sem marca d'água) if (targetUrl.includes('tiktok.com') || targetUrl.includes('douyin.com')) { try { const tikRes = await fetch(`https://www.tikwm.com/api/?url=${encodeURIComponent(targetUrl)}`, { headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' } }); const tikData = await tikRes.json(); if (tikData && tikData.code === 0 && tikData.data && (tikData.data.play || tikData.data.wmplay)) { return res.status(200).json({ success: true, downloadUrl: tikData.data.play || tikData.data.wmplay }); } } catch (e) {} } // 2. Cascata de instâncias universais (Instagram, Facebook, YouTube, Kwai, X, TikTok) const instances = [ { url: 'https://cobalt-api.kwiatekmiki.com', v10: true }, { url: 'https://cobalt.api.timelessnesses.me', v10: true }, { url: 'https://api.cobalt.blackcat.sweeux.org', v10: true }, { url: 'https://co.wuk.sh/api/json', v10: false } ]; for (const inst of instances) { try { const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 7000); const payload = inst.v10 ? { url: targetUrl, videoQuality: '720', filenameStyle: 'classic' } : { url: targetUrl, vQuality: '720', filenamePattern: 'classic' }; const response = await fetch(inst.url, { method: 'POST', headers: { 'Accept': 'application/json', 'Content-Type': 'application/json', 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' }, body: JSON.stringify(payload), signal: controller.signal }); clearTimeout(timeout); if (response.ok) { const data = await response.json(); const videoUrl = data.url || (data.picker && data.picker[0] && data.picker[0].url); if (videoUrl) { return res.status(200).json({ success: true, downloadUrl: videoUrl }); } } } catch (e) { // Passa automaticamente para o próximo servidor caso este falhe } } // 3. Se for um link direto de arquivo (.mp4, .webm) if (/\.(mp4|webm|mov)(\?.*)?$/i.test(targetUrl)) { return res.status(200).json({ success: true, downloadUrl: targetUrl }); } return res.status(500).json({ success: false, error: 'O servidor da rede social bloqueou a extração automática deste link. Tente outro link.' }); }
